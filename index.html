Data URL/base64 besar tidak ditaruh di sel Sheets. // Untuk foto besar, gunakan tautan Google Drive pada field attachments. if (Array.isArray(jsonSafe.attachments)) { jsonSafe.attachments = jsonSafe.attachments.map(function(a) { if (typeof a === 'string' && a.startsWith('data:')) return '[Foto base64 tidak disimpan di Sheet; gunakan URL Drive]'; if (a && typeof a === 'object') { const copy = Object.assign({}, a); ['dataUrl', 'base64', 'data'].forEach(function(k) { if (typeof copy[k] === 'string' && copy[k].startsWith('data:')) copy[k] = '[base64 tidak disimpan]'; }); return copy; } return a; }); } const fullJson = JSON.stringify(jsonSafe); const row = [ id, now, payload.status || (recordType === 'Listing' ? 'Draft' : 'Baru'), payload.name || payload.nama || payload.ownerName || '', payload.phone || payload.whatsapp || payload.contact || '', payload.type || payload.submissionType || '', payload.category || payload.kategori || '', payload.title || payload.judul || payload.name || '', payload.location || payload.lokasi || payload.address || '', payload.price || payload.harga || '', fullJson.length > 49000 ? fullJson.slice(0, 49000) : fullJson ]; if (rowNumber > 0) sheet.getRange(rowNumber, 1, 1, row.length).setValues([row]); else sheet.appendRow(row); } if (payload.clientRequestId) { PropertiesService.getScriptProperties().setProperty('ack_' + payload.clientRequestId, new Date().toISOString()); } return ContentService.createTextOutput(JSON.stringify({ok:true})).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ok:false,error:String(err)})).setMimeType(ContentService.MimeType.JSON); } } function getSheet_(name) { const ss = SpreadsheetApp.openById(SPREADSHEET_ID); let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); if (sh.getLastRow() === 0) sh.appendRow(SHEET_HEADERS); return sh; } function readRecords_(name) { const sh = getSheet_(name); const values = sh.getDataRange().getValues(); const out = []; for (let i = 1; i < values.length; i++) { const raw = values[i][10]; if (!raw) continue; try { out.push(JSON.parse(raw)); } catch (_) {} } return out; } function authorizeSpreadsheetAccess() { getSheet_('Formulir'); getSheet_('Listing'); }
